async function answerUser(text) {
  const local = classifyLocally(text);
  if (local.action) return applyLocalAction(local);

  try {
    return await puter.ai.chat(buildPrompt(text), { model: "grok-beta" });
  } catch (error) {
    return localFallback(text);
  }
}

chance final

base + dificuldade A dificuldade não "quebra" o jogo. Ela ajusta a probabilidade para ficar mais fácil ou mais difícil. 
function calcWinChance(base) {
  if (WIN_DIFFICULTY <= 0) return 0;
  if (WIN_DIFFICULTY >= 1) return 1;

  const adjustment = (WIN_DIFFICULTY - 0.5) * 2;
  return base + adjustment * (1 - base) * 0.8;
}
